\begin{frame}{Current System Challenges}
    \begin{itemize}
        \item \textbf{Scalability:} Monolithic services are becoming a bottleneck during high traffic.
        \item \textbf{Data Isolation:} Stale reads happening due to tightly coupled data layers.
        \item \textbf{Authentication:} Moving from stateful sessions to stateless JWT tokens (OAuth 2.0).
    \end{itemize}
    \vspace{1em}
    \textit{Goal:} Decompose the core services to improve independent scaling capabilities and maintainability.
\end{frame}
